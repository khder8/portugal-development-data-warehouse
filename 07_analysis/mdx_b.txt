-- Task 7B: Portugal compared with the Western European countries

-- Query B1: GDP per capita by decade, Portugal and the Western European countries. Cube: Economy
SELECT
  [Time].[Time Hierarchy].[Decade].Members ON COLUMNS,
  {[Country].[Country Hierarchy].[Focus].[Portugal],
   [Country].[Country Hierarchy].[Western Europe].Children} ON ROWS
FROM [Economy]
WHERE ([Measures].[GDP per Capita])

-- Query B2: GDP per capita, GNI per capita and life expectancy in 2022. Virtual cube: Economy and Society
SELECT
  {[Measures].[GDP per Capita], [Measures].[GNI per Capita], [Measures].[Life Expectancy]} ON COLUMNS,
  {[Country].[Country Hierarchy].[Focus].[Portugal],
   [Country].[Country Hierarchy].[Western Europe].Children} ON ROWS
FROM [Economy and Society]
WHERE ([Time].[Time Hierarchy].[2020].[2022])
