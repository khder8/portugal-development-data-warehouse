-- Task 7A: evolution of Portugal's GDP per capita since 1900 and growth by decade
-- Calculated member with PrevMember on the Decade level. Cube: Economy
WITH MEMBER [Measures].[Decade growth %] AS
  IIF(IsEmpty(([Measures].[GDP per Capita], [Time].[Time Hierarchy].CurrentMember.PrevMember)), NULL,
      [Measures].[GDP per Capita] / ([Measures].[GDP per Capita], [Time].[Time Hierarchy].CurrentMember.PrevMember) - 1),
  FORMAT_STRING = '0.0%'
SELECT {[Measures].[GDP per Capita], [Measures].[Decade growth %]} ON COLUMNS,
  [Time].[Time Hierarchy].[Decade].Members ON ROWS
FROM [Economy]
WHERE ([Country].[Country Hierarchy].[Focus].[Portugal])
